import {initializeApp} from 'https://www.gstatic.com/firebasejs/11.10.0/firebase-app.js';
import {getAuth,onAuthStateChanged,createUserWithEmailAndPassword,signInWithEmailAndPassword,signOut,updateProfile} from 'https://www.gstatic.com/firebasejs/11.10.0/firebase-auth.js';
import {getFirestore,collection,addDoc,getDocs,query,where,orderBy,doc,getDoc,serverTimestamp} from 'https://www.gstatic.com/firebasejs/11.10.0/firebase-firestore.js';
import {firebaseConfig} from './firebase-config.js';
const $=id=>document.getElementById(id), msg=(t,bad=false)=>{ $('message').textContent=t; $('message').className=bad?'error':'ok';};
const app=initializeApp(firebaseConfig),auth=getAuth(app),db=getFirestore(app);
let user=null,isAdmin=false,questions=[],activeTest='';
$('year').textContent=new Date().getFullYear();
function node(tag,txt,cls){const n=document.createElement(tag);if(txt!==undefined)n.textContent=txt;if(cls)n.className=cls;return n;}
function show(page){if(page==='admin'&&!isAdmin){msg('एडमिन की अनुमति नहीं है।',true);return;}document.querySelectorAll('main>section').forEach(s=>s.classList.toggle('hide',s.id!==page));msg('');if(page==='notes')loadNotes();if(page==='tests')loadTests();if(page==='results')loadResults();}
document.querySelectorAll('[data-page]').forEach(b=>b.addEventListener('click',()=>show(b.dataset.page)));
const fail=e=>msg(e?.message||'कुछ गलत हुआ।',true);
onAuthStateChanged(auth,async u=>{user=u;isAdmin=false;if(u){try{const snap=await getDoc(doc(db,'admins',u.uid));isAdmin=snap.exists();}catch(e){fail(e);}}$('loggedOut').classList.toggle('hide',!!u);$('loggedIn').classList.toggle('hide',!u);$('userInfo').textContent=u?`${u.displayName||'विद्यार्थी'} — ${u.email}`:'';$('accountNav').textContent=u?'👤 मेरा अकाउंट':'🔐 लॉगिन';$('adminNav').classList.toggle('hide',!isAdmin);if(!isAdmin&&!$('admin').classList.contains('hide'))show('home');});
$('signupForm').addEventListener('submit',async e=>{e.preventDefault();try{const cred=await createUserWithEmailAndPassword(auth,$('signupEmail').value.trim(),$('signupPassword').value);await updateProfile(cred.user,{displayName:$('signupName').value.trim()});msg('अकाउंट बन गया!');show('account');}catch(err){fail(err);}});
$('loginForm').addEventListener('submit',async e=>{e.preventDefault();try{await signInWithEmailAndPassword(auth,$('loginEmail').value.trim(),$('loginPassword').value);msg('लॉगिन सफल!');show('account');}catch(err){fail(err);}});
$('logoutBtn').onclick=async()=>{await signOut(auth);show('home');};
async function loadNotes(){const list=$('notesList');list.replaceChildren(node('p','लोड हो रहा है...'));try{const snaps=await getDocs(collection(db,'notes'));list.replaceChildren();if(snaps.empty)list.append(node('p','अभी कोई नोट नहीं जोड़ा गया है।'));snaps.forEach(d=>{const x=d.data(),card=node('article',undefined,'card');card.append(node('span',x.subject||'सामान्य','badge'),node('h3',x.title||''),node('p',x.body||''));if(x.url){try{const url=new URL(x.url);if(url.protocol==='https:'){const a=node('a','PDF / सामग्री खोलें');a.href=url.href;a.target='_blank';a.rel='noopener noreferrer';card.append(a);}}catch{}}list.append(card);});}catch(e){fail(e);}}
async function loadTests(){const list=$('testList');$('quiz').classList.add('hide');list.replaceChildren(node('p','लोड हो रहा है...'));try{const snaps=await getDocs(collection(db,'questions'));questions=snaps.docs.map(d=>({id:d.id,...d.data()}));const groups=Object.groupBy(questions,q=>q.test||'सामान्य');list.replaceChildren();if(!questions.length)list.append(node('p','अभी कोई टेस्ट नहीं है। एडमिन प्रश्न जोड़ सकता है।'));Object.entries(groups).forEach(([title,qs])=>{const card=node('article',undefined,'card');card.append(node('h3',title),node('p',`${qs.length} प्रश्न`));const btn=node('button','टेस्ट शुरू करें');btn.onclick=()=>startQuiz(title);card.append(btn);list.append(card);});}catch(e){fail(e);}}
function startQuiz(title){activeTest=title;const qs=questions.filter(q=>q.test===title);const box=$('quiz');box.replaceChildren(node('h2',title));const form=node('form');qs.forEach((q,i)=>{const area=node('fieldset',undefined,'card');area.append(node('legend',`${i+1}. ${q.text}`));(q.options||[]).forEach((opt,j)=>{const label=node('label',undefined,'choice'),input=node('input');input.type='radio';input.name=`question${i}`;input.value=String(j);label.append(input,document.createTextNode(`${'ABCD'[j]}. ${opt}`));area.append(label);});form.append(area);});const submit=node('button','टेस्ट जमा करें');form.append(submit);form.onsubmit=async e=>{e.preventDefault();if(!user){msg('रिजल्ट सेव करने के लिए पहले लॉगिन करें।',true);show('account');return;}const answers=qs.map((_,i)=>{const v=new FormData(form).get(`question${i}`);return v===null?null:Number(v);});if(answers.some(a=>a===null)){msg('सभी प्रश्नों के उत्तर दें।',true);return;}const score=qs.reduce((s,q,i)=>s+(answers[i]===Number(q.correct)?1:0),0);try{await addDoc(collection(db,'results'),{uid:user.uid,test:title,score,total:qs.length,createdAt:serverTimestamp()});box.replaceChildren(node('h2',`आपका स्कोर: ${score}/${qs.length}`));qs.forEach((q,i)=>{const p=node('p',`${i+1}. ${q.text} — आपका उत्तर: ${'ABCD'[answers[i]]}; सही उत्तर: ${'ABCD'[q.correct]}`);p.className=answers[i]===Number(q.correct)?'ok':'error';box.append(p);});msg('रिजल्ट सेव हो गया!');}catch(err){fail(err);}};box.append(form);box.classList.remove('hide');box.scrollIntoView({behavior:'smooth'});}
async function loadResults(){const list=$('resultList');list.replaceChildren();if(!user){list.append(node('p','रिजल्ट देखने के लिए लॉगिन करें।'));return;}try{const snaps=await getDocs(query(collection(db,'results'),where('uid','==',user.uid)));const items=snaps.docs.map(d=>d.data()).sort((a,b)=>(b.createdAt?.seconds||0)-(a.createdAt?.seconds||0));if(!items.length)list.append(node('p','अभी कोई रिजल्ट नहीं है।'));items.forEach(r=>{const c=node('div',undefined,'card');c.append(node('h3',r.test),node('p',`${r.score}/${r.total} — ${Math.round(100*r.score/r.total)}%`));list.append(c);});}catch(e){fail(e);}}
$('noteForm').onsubmit=async e=>{e.preventDefault();if(!isAdmin)return;try{await addDoc(collection(db,'notes'),{subject:$('noteSubject').value.trim(),title:$('noteTitle').value.trim(),body:$('noteBody').value.trim(),url:$('noteUrl').value.trim(),createdAt:serverTimestamp()});e.target.reset();msg('नोट प्रकाशित हो गया।');}catch(err){fail(err);}};
$('questionForm').onsubmit=async e=>{e.preventDefault();if(!isAdmin)return;try{await addDoc(collection(db,'questions'),{test:$('qTest').value.trim(),text:$('qText').value.trim(),options:[$('qA').value,$('qB').value,$('qC').value,$('qD').value],correct:Number($('qCorrect').value),createdAt:serverTimestamp()});e.target.reset();msg('प्रश्न जोड़ दिया गया।');}catch(err){fail(err);}};
